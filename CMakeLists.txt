cmake_minimum_required(VERSION 3.18)
project(cpymacs VERSION 0.1.1 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(CPYMACS_GUI "Build the independent C/X11 GUI frontend" ON)
option(CPYMACS_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(CPYMACS_LTO "Enable link-time optimization when available" ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c>=0.15)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter Development.Embed)
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion -Wno-misleading-indentation)
if(CPYMACS_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(CPYMACS_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(ipo_supported)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
add_library(cpymacs_text STATIC src/text.c)
target_include_directories(cpymacs_text PUBLIC include)
add_library(cpymacs_transport STATIC src/transport.c)
target_include_directories(cpymacs_transport PUBLIC include)
target_link_libraries(cpymacs_transport PUBLIC PkgConfig::JSONC)
add_library(cpymacs_core STATIC src/buffer.c src/commands.c src/syntax.c src/python.c src/view.c src/protocol.c)
target_include_directories(cpymacs_core PUBLIC include)
target_link_libraries(cpymacs_core PUBLIC cpymacs_text cpymacs_transport PRIVATE Python3::Python)
add_library(cpymacs_term_style STATIC src/term_style.c)
target_include_directories(cpymacs_term_style PUBLIC include)
target_link_libraries(cpymacs_term_style PUBLIC m)
add_executable(cpymacs src/main.c src/tui.c)
target_link_libraries(cpymacs PRIVATE cpymacs_core cpymacs_term_style Python3::Python)
target_compile_definitions(cpymacs PRIVATE
  CPYMACS_PYTHON_DIR="${CMAKE_CURRENT_SOURCE_DIR}/python"
  CPYMACS_INSTALL_DATA="${CMAKE_INSTALL_FULL_DATADIR}/cpymacs/python")
if(CPYMACS_GUI)
  find_package(X11 REQUIRED)
  add_executable(cpymacs-gui src/gui_x11.c)
  target_include_directories(cpymacs-gui PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(cpymacs-gui PRIVATE cpymacs_transport ${X11_LIBRARIES})
  install(TARGETS cpymacs-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS cpymacs RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY python/ DESTINATION ${CMAKE_INSTALL_DATADIR}/cpymacs/python
        PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES LICENSE README.md CHANGELOG.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/cpymacs/examples)
if(BUILD_TESTING)
  add_executable(test_term_style tests/test_term_style.c)
  target_link_libraries(test_term_style PRIVATE cpymacs_term_style)
  add_test(NAME terminal_palette COMMAND test_term_style)
  add_test(NAME terminal_colors COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_terminal_colors.py $<TARGET_FILE:cpymacs>)
  set_tests_properties(terminal_colors PROPERTIES TIMEOUT 90)
  if(CPYMACS_GUI)
    add_executable(gui_driver tests/gui_driver.c)
    target_include_directories(gui_driver PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(gui_driver PRIVATE ${X11_LIBRARIES})
    add_test(NAME frontends COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_frontends.py $<TARGET_FILE:cpymacs> $<TARGET_FILE:gui_driver>)
  else()
    add_test(NAME frontends COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_frontends.py $<TARGET_FILE:cpymacs>)
  endif()
  set_tests_properties(frontends PROPERTIES TIMEOUT 90)
  add_executable(test_text tests/test_text.c)
  target_link_libraries(test_text PRIVATE cpymacs_text)
  add_test(NAME text_properties COMMAND test_text)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:cpymacs>)
  set_tests_properties(integration PROPERTIES TIMEOUT 120 ENVIRONMENT "CPYMACS_PYTHON_DIR=${CMAKE_CURRENT_SOURCE_DIR}/python")
endif()
add_executable(bench_text bench/bench_text.c)
target_link_libraries(bench_text PRIVATE cpymacs_text)
